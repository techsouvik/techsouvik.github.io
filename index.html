from typing import TypedDict, Annotated, List
from langgraph.graph import StateGraph, END
import operator

class CoordinatorState(TypedDict):
    audit_findings: Annotated[List[dict], operator.add]
    threat_level: str

def route_next_agent(state: CoordinatorState) -> str:
    return "cve_scanner" if state["threat_level"] == "CRITICAL" else "ast_analyzer"

graph = StateGraph(CoordinatorState)
graph.add_node("coordinator", triage_node)
graph.add_node("ast_analyzer", static_ast_node)
graph.add_node("cve_scanner", cve_scanner_node)
graph.add_conditional_edges("coordinator", route_next_agent)
app = graph.compile()
